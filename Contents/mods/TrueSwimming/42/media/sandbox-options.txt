/* XP multiplier of the Swimming skill. The engine (IsoGameCharacter$XP.AddXP, 42.21) looks up the sandbox option
   "MultiplierConfig.<skill>" whenever the global XP multiplier toggle is OFF, but it only creates that option for
   vanilla skills: a custom skill then throws a NullPointerException on every XP gain. Declaring the option here with
   that exact id registers it under the same name, so the lookup succeeds. The vanilla XP multipliers are listed at the
   end of the sandbox page named "Character" (ServerSettingsScreen.lua), so this page name puts the slider right after them. */
option MultiplierConfig.Swimming
{
	type = double,
	min = 0.0,
	max = 1000.0,
	default = 1.0,
	page = Character,
	translation = MultiplierConfig_Swimming,
}
